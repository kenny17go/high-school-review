-- 高中段考複習平台 V4.2 / Supabase schema
-- 請在 Supabase Dashboard > SQL Editor 執行。
-- 不要把 service_role key 放進前端。

create table if not exists public.schools (
  id bigint generated by default as identity primary key,
  name text not null unique,
  source_status text,
  source_tone text check (source_tone in ('good','mid','low')),
  description text,
  official_url text,
  created_at timestamptz not null default now()
);

create table if not exists public.exams (
  id bigint generated by default as identity primary key,
  school_id bigint references public.schools(id) on delete cascade,
  academic_year int not null,
  term int not null check (term in (1,2)),
  exam_name text not null,
  grade int not null default 1,
  subject text not null default '數學',
  scope_text text,
  source_url text,
  source_status text,
  created_at timestamptz not null default now()
);

create table if not exists public.questions (
  id bigint primary key,
  school_id bigint references public.schools(id) on delete set null,
  exam_id bigint references public.exams(id) on delete set null,
  topic text not null,
  subtopic text,
  difficulty text not null check (difficulty in ('基礎','中等','挑戰')),
  question_text text not null,
  options jsonb not null,
  correct_index int not null check (correct_index between 0 and 3),
  explanation text not null,
  is_original boolean not null default true,
  source_url text,
  created_at timestamptz not null default now()
);

create table if not exists public.attempts (
  id bigint generated by default as identity primary key,
  user_id uuid not null references auth.users(id) on delete cascade,
  question_id bigint not null references public.questions(id) on delete cascade,
  selected_index int,
  is_correct boolean not null,
  mode text not null default 'practice',
  school_name text,
  academic_year int,
  exam_name text,
  attempted_at timestamptz not null default now()
);

create table if not exists public.wrong_questions (
  id bigint generated by default as identity primary key,
  user_id uuid not null references auth.users(id) on delete cascade,
  question_id bigint not null references public.questions(id) on delete cascade,
  wrong_count int not null default 1,
  last_wrong_at timestamptz not null default now(),
  mastered boolean not null default false,
  unique(user_id, question_id)
);

create index if not exists exams_school_idx on public.exams(school_id);
create index if not exists questions_topic_idx on public.questions(topic);
create index if not exists questions_school_idx on public.questions(school_id);
create index if not exists attempts_user_idx on public.attempts(user_id);
create index if not exists attempts_question_idx on public.attempts(question_id);
create index if not exists wrong_questions_user_idx on public.wrong_questions(user_id);

alter table public.schools enable row level security;
alter table public.exams enable row level security;
alter table public.questions enable row level security;
alter table public.attempts enable row level security;
alter table public.wrong_questions enable row level security;

-- 公開教材資料：任何訪客都可讀；前端不能改。
revoke all on table public.schools, public.exams, public.questions from anon, authenticated;
grant select on table public.schools, public.exams, public.questions to anon, authenticated;

drop policy if exists "public read schools" on public.schools;
create policy "public read schools" on public.schools for select to anon, authenticated using (true);

drop policy if exists "public read exams" on public.exams;
create policy "public read exams" on public.exams for select to anon, authenticated using (true);

drop policy if exists "public read questions" on public.questions;
create policy "public read questions" on public.questions for select to anon, authenticated using (true);

-- 個人作答資料：只有登入者可存取自己的資料。
revoke all on table public.attempts, public.wrong_questions from anon, authenticated;
grant select, insert, update, delete on table public.attempts, public.wrong_questions to authenticated;

drop policy if exists "read own attempts" on public.attempts;
create policy "read own attempts" on public.attempts for select to authenticated
using ((select auth.uid()) = user_id);

drop policy if exists "insert own attempts" on public.attempts;
create policy "insert own attempts" on public.attempts for insert to authenticated
with check ((select auth.uid()) = user_id);

drop policy if exists "delete own attempts" on public.attempts;
create policy "delete own attempts" on public.attempts for delete to authenticated
using ((select auth.uid()) = user_id);

drop policy if exists "read own wrong questions" on public.wrong_questions;
create policy "read own wrong questions" on public.wrong_questions for select to authenticated
using ((select auth.uid()) = user_id);

drop policy if exists "insert own wrong questions" on public.wrong_questions;
create policy "insert own wrong questions" on public.wrong_questions for insert to authenticated
with check ((select auth.uid()) = user_id);

drop policy if exists "update own wrong questions" on public.wrong_questions;
create policy "update own wrong questions" on public.wrong_questions for update to authenticated
using ((select auth.uid()) = user_id)
with check ((select auth.uid()) = user_id);

drop policy if exists "delete own wrong questions" on public.wrong_questions;
create policy "delete own wrong questions" on public.wrong_questions for delete to authenticated
using ((select auth.uid()) = user_id);
